while true; do
    timestamp=$(date "+%Y-%m-%d %H:%M:%S")
    result=$(curl -s -o /dev/null \
             -w "%{http_code} %{time_total} " \
             --proxy http://n5-testing4:P%40ssw0rd1234@192.168.52.236:8080 \

    sleep 1
done

screen -S 41a-ping

ping -D 192.168.52.236 | while read line; do
  if [[ "$line" =~ \[([0-9]+\.[0-9]+)\] ]]; then
    ts=${BASH_REMATCH[1]}
    human=$(TZ=Asia/Shanghai date -d "@$ts" "+%F %T.%6N")
    echo "[$human] ${line#*\] }"
  else
    echo "$line"
  fi
done >> /tmp/41a-ping.txt 2>&1
